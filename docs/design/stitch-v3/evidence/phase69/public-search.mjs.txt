import {createRequire} from 'node:module';
const require=createRequire('/workspace/yieldloom-redesign-phase32/package.json');const {chromium}=require('playwright');
const hosts=new Set(['yieldloom-app.vercel.app','totobubu.github.io','fonts.googleapis.com','fonts.gstatic.com','www.gstatic.com','unpkg.com']);
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});
try{for(const [name,base] of [['vercel','https://yieldloom-app.vercel.app/'],['pages','https://totobubu.github.io/yieldloom-app/']]){
 const context=await browser.newContext({viewport:{width:1440,height:1000}});const page=await context.newPage();let verifiedFetches=0,blockedWrites=0;const errors=[],excludedHosts=new Set(),failedHosts=new Set();page.on('pageerror',e=>errors.push(e.message));
 await context.route('**/*',async route=>{const req=route.request(),url=new URL(req.url());if(!['GET','HEAD','OPTIONS'].includes(req.method())){blockedWrites++;return route.abort()}if(!hosts.has(url.hostname)||/\/api\/(family-portfolio|toss|sheet-ledger|portfolio-capital|admin)/.test(url.pathname)){excludedHosts.add(url.hostname);return route.abort()}
  try{const res=await fetch(url,{method:req.method(),signal:AbortSignal.timeout(25000)});if(!hosts.has(new URL(res.url).hostname))throw Error('Unexpected redirect');verifiedFetches++;const headers=Object.fromEntries([...res.headers].filter(([k])=>!['content-encoding','content-length','transfer-encoding','set-cookie'].includes(k)));await route.fulfill({status:res.status,headers,body:Buffer.from(await res.arrayBuffer())})}catch{failedHosts.add(url.hostname);await route.abort()}});
 try{const res=await page.goto(base+'?release='+process.env.PUBLIC_RELEASE_SHA,{waitUntil:'networkidle',timeout:45000});await page.locator('h1').first().waitFor();await page.evaluate(()=>document.fonts.ready);const desktop=await page.evaluate(()=>({title:document.title,headings:[...document.querySelectorAll('h1')].map(e=>e.textContent),textLength:document.body.innerText.length,overflow:document.documentElement.scrollWidth>innerWidth}));await page.screenshot({path:'/tmp/phase69-production-search-'+name+'-1440.png'});await page.setViewportSize({width:390,height:844});await page.screenshot({path:'/tmp/phase69-production-search-'+name+'-390.png'});const mobileOverflow=await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth);console.log(JSON.stringify({name,status:res.status(),desktop,mobileOverflow,verifiedFetches,blockedWrites,excludedHosts:[...excludedHosts],failedHosts:[...failedHosts],errors,transport:'Anonymous public HTTPS fetched using Node inherited CA verification; direct Chromium TLS remains blocked'}));}catch(e){console.log(JSON.stringify({name,error:e.message.split('\n')[0],verifiedFetches,blockedWrites,failedHosts:[...failedHosts]}))}await context.close();}}
finally{await browser.close()}
