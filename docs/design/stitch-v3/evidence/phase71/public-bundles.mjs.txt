import fs from 'node:fs';
import {createHash} from 'node:crypto';
import assert from 'node:assert/strict';

 assert.ok(Object.values(styleChecks).every(Boolean),JSON.stringify(styleChecks));

 assert.ok(/\.generic-review-table\[data-v-[^\]]+\] \.p-datatable-paginator-bottom\{border:0\}/.test(css),'review footer duplicate border removed');assert.ok(css.includes('flex:0 1 auto'),'reference code flex');console.log(JSON.stringify({base,scope:'new deployed review styles',footerBorderRemoved:true,referenceCodeFlex:true}));
 const receiptChecks={scopedFont:/@media\(min-width:641px\)\{\.receipt-refresh-copy\[data-v-[^\]]+\]\{font-family:YieldloomReceiptReference,sans-serif\}/.test(css),heading:content.includes('receipt-next-heading'),scopeLabel:content.includes('선택 계좌 전체 내역 CSV로 내려받기')};assert.ok(Object.values(receiptChecks).every(Boolean),JSON.stringify(receiptChecks));
 const manifest=JSON.parse(fs.readFileSync('src/assets/fonts/stitch-receipt/SOURCE.json','utf8'));const refs=[...css.matchAll(/url\(([^)]+Pretendard-(Regular|Bold)[^)]+)\)/g)];assert.equal(refs.length,2);const fontProof=[];for(const [,raw,name] of refs){const u=new URL(raw.replace(/^["']|["']$/g,''),new URL(cssPath,base));assert.equal(u.origin,new URL(base).origin);const res=await fetch(u);assert.equal(res.status,200);assert.equal(new URL(res.url).origin,u.origin);const data=Buffer.from(await res.arrayBuffer());assert.equal(data.subarray(0,4).toString('ascii'),'wOF2');const expected=manifest.fonts.find(f=>f.file==='Pretendard-'+name+'.woff2');const hash=createHash('sha256').update(data).digest('hex');assert.equal(hash,expected.sha256);assert.equal(data.length,expected.bytes);fontProof.push({path:u.pathname,bytes:data.length,sha256:hash,contentType:res.headers.get('content-type')})}console.log(JSON.stringify({base,scope:'new deployed receipt CSS and public static fonts only; authenticated receipt not exercised',checks:receiptChecks,fonts:fontProof}));

 const checks={rawPreview:content.includes('원본 데이터 미리보기'),mobilePicker:content.includes('upload-mobile-picker'),accountConfirmation:content.includes('upload-account-confirmation'),reviewPresent:content.includes('import-review-agreement'),reviewRowEditor:content.includes('review-row-editor'),reviewRawUnsupported:content.includes('원문 보기 (미지원)'),reviewSaveCaption:content.includes('배당달력 포트폴리오에 저장 및 반영'),ledgerPresent:content.includes('ledger-records'),ledgerGates:content.includes('계좌 XIRR 결과')};
