import {createRequire} from 'node:module';
import fs from 'node:fs/promises';
const repo='/workspace/yieldloom-redesign-phase70';
const {chromium}=createRequire(repo+'/package.json')('playwright');
const {installStaticAssets}=await import(repo+'/scripts/design/browser-static-assets.mjs');
const inventory=JSON.parse(await fs.readFile('/tmp/phase71-font-reference-inventory.json','utf8'));
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});
const results=[];
try {for(const f of inventory.frames.filter(f=>!f.providedPretendardCdn)) {
 const context=await browser.newContext({viewport:{width:f.device==='Mobile'?390:834,height:1100}});
 const assetFailures=await installStaticAssets(context);
 const page=await context.newPage();const responses=[];
 page.on('response',r=>{if(r.url().includes('pretendard')||r.url().includes('fonts.googleapis.com/css2?family=Inter,'))responses.push({url:r.url(),status:r.status()})});
 const html=await fs.readFile(repo+'/docs/design/stitch-v3/originals/17382728461495959991/'+f.screen+'.html','utf8');
 await page.route('http://127.0.0.1:5203/phase71-original',r=>r.fulfill({contentType:'text/html',body:html}));
 await page.goto('http://127.0.0.1:5203/phase71-original');await page.evaluate(()=>document.fonts.ready);
 const samples=await page.evaluate(()=>[...document.querySelectorAll('h1,h2,h3,p')].filter(e=>/[가-힣]/.test(e.textContent)).slice(0,4).map((e,i)=>{e.setAttribute('data-font-audit',String(i));const s=getComputedStyle(e);return {index:i,tag:e.tagName,text:e.textContent.trim().slice(0,55),declared:s.fontFamily,font:s.fontSize,line:s.lineHeight}}));
 const cdp=await context.newCDPSession(page);await cdp.send('DOM.enable');await cdp.send('CSS.enable');const {root}=await cdp.send('DOM.getDocument');
 for(const s of samples){const {nodeId}=await cdp.send('DOM.querySelector',{nodeId:root.nodeId,selector:`[data-font-audit="${s.index}"]`});s.actual=(await cdp.send('CSS.getPlatformFontsForNode',{nodeId})).fonts}
 await cdp.detach();results.push({screen:f.screen,family:f.family,device:f.device,responses,assetFailures,samples,scope:'preserved source HTML with inherited-CA static asset bridge; CDN declaration is not proof of selected font; native PNG not verified'});
 await context.close();
}} finally {await browser.close()}
await fs.writeFile('/tmp/phase71-reference-font-other-audit.json',JSON.stringify(results,null,2)+'\n');
for(const r of results)console.log(JSON.stringify({screen:r.screen,family:r.family,device:r.device,assetFailures:r.assetFailures,samples:r.samples.map(s=>({tag:s.tag,declared:s.declared,actual:s.actual.map(f=>({family:f.familyName,custom:f.isCustomFont}))}))}));
