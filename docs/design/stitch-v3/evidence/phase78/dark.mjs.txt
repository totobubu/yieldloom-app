import {createRequire} from 'node:module';import fs from 'node:fs/promises';import assert from 'node:assert/strict';
import {installStitchFixtures} from '/workspace/yieldloom-redesign-phase78/tests/stitch-fixtures.mjs';import {installStaticAssets} from '/workspace/yieldloom-redesign-phase78/scripts/design/browser-static-assets.mjs';
const {chromium}=createRequire('/workspace/yieldloom-redesign-phase78/package.json')('playwright');const b=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});const rows=[];
try{for(const width of [1280,768,390]){const p=await b.newPage({viewport:{width,height:1024},deviceScaleFactor:2});await installStitchFixtures(p);await installStaticAssets(p);const errors=[];p.on('pageerror',e=>errors.push(e.message));await p.goto('http://127.0.0.1:5278/stock/DEMO-A?tab=dividends');const t=p.locator('.dividend-trend');await t.locator('.month-bar').last().waitFor();await p.locator('.fullscreen-data-loading-mask').waitFor({state:'hidden'});for(const theme of ['light','dark']){if(theme==='dark')await p.getByRole('button',{name:'다크 모드',exact:true}).click();await t.locator('.month-bar').last().focus();await p.keyboard.press('Enter');await t.screenshot({path:`/tmp/phase78/after/status-${width}-${theme}.png`,animations:'disabled'});const state=await t.evaluate(e=>({overflow:document.documentElement.scrollWidth>innerWidth+1,plotWidth:e.querySelector('.trend-plot').getBoundingClientRect().width,scrollLeft:e.querySelector('.trend-chart').scrollLeft,missing:e.querySelector('.month-bar:last-child .month-status')?.textContent,partial:e.querySelector('.month-bar:nth-of-type(11) .month-status')?.textContent,background:getComputedStyle(e.querySelector('.trend-plot')).backgroundColor,text:getComputedStyle(e.querySelector('.month-bar')).color}));assert.equal(state.overflow,false);assert.equal(state.missing,'미확인');assert.equal(state.partial,'부분');assert.deepEqual(errors,[]);rows.push({width,theme,...state,errors});}await p.close()}}finally{await b.close();await fs.writeFile('/tmp/phase78/after/themes.json',JSON.stringify(rows,null,2)+'\n')}console.log('PASS light/dark, keyboard-revealed status, missing/partial, no overflow/errors: 6 states');
