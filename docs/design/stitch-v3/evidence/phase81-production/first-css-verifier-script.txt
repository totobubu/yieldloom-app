import assert from 'node:assert/strict';
import fs from 'node:fs/promises';
import {createHash} from 'node:crypto';
const expected=process.env.PUBLIC_RELEASE_SHA;assert.match(expected,/^[a-f0-9]{40}$/);
const built=JSON.parse(await fs.readFile('/tmp/phase81/css-build-proof.json','utf8'));const rows=[];
for(const base of ['https://yieldloom-app.vercel.app/','https://totobubu.github.io/yieldloom-app/']){
 const responses=[];
 const get=async input=>{const url=new URL(input,base);assert.equal(url.hostname,new URL(base).hostname);const r=await fetch(url,{cache:'no-store',headers:{'Cache-Control':'no-cache'},signal:AbortSignal.timeout(30000)});assert.equal(r.status,200);responses.push({path:url.pathname,status:r.status});return r;};
 const release=await(await get('app-release.json?css='+Date.now())).json();assert.equal(release.commit,expected);assert.equal(release.metadataSha256,'aba948fd05cec02755347f8437f769b394802ecc4fef37e24f9127e85e083e7e');
 const html=await(await get('?css='+expected)).text();const match=html.match(/<script[^>]*type="module"[^>]*src="([^"]+)"/);assert.ok(match,'actual current index must load module entry');
 const entry=await(await get(match[1])).text();assert.ok(entry.includes(built.path.split('/').at(-1)),'current entry must reference expected FamilyPortfolio CSS');
 const css=Buffer.from(await(await get(built.path)).arrayBuffer());const digest=createHash('sha256').update(css).digest('hex');assert.equal(digest,built.sha256);assert.ok(css.toString().includes(built.rule));
 rows.push({base,commit:release.commit,metadataSha256:release.metadataSha256,cssPath:built.path,sha256:digest,entryReferencesExpectedCss:true,expectedRule:built.rule,responses});
}
await fs.writeFile('/tmp/phase81/public-css-verification.json',JSON.stringify({checkedAt:new Date().toISOString(),scope:'Actual public release/index/entry and static CSS delivery only. No authenticated account rendering, private API, customer PDF or DB-write verification. No finance data substitution.',rows},null,2)+'\n');console.log('Verified both actual releases and current entry references to exact built CSS; authenticated account flow remains unverified.');
